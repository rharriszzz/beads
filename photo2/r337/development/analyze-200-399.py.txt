def translations(ab,c,blocks,limit=40):
    occ={tuple(v):i for i,v in enumerate(ab)};out=[]
    for x in range(-limit,limit+1):
        for y in range(-limit,limit+1):
            if x+y<0 or (x+y==0 and x<=0):continue
            pairs=[(i,occ[tuple(v+[x,y])]) for i,v in enumerate(ab) if tuple(v+[x,y]) in occ]
            if len(pairs)<25:continue
            ii,jj=np.array(pairs).T;far=np.minimum(abs(blocks[ii]-blocks[jj]),12-abs(blocks[ii]-blocks[jj]))>=2
            dev=~np.isin(blocks[ii],[2,5,8,11])&~np.isin(blocks[jj],[2,5,8,11]);test=~dev
            conf=np.bincount(c[ii]*3+c[jj],minlength=9).reshape(3,3)
            perms=list(itertools.permutations(range(3)));best=max(perms,key=lambda p:sum(conf[j,p[j]] for j in range(3)))
            out.append(dict(vector=[x,y],pairs=len(ii),agree=int((c[ii]==c[jj]).sum()),agreement=float((c[ii]==c[jj]).mean()),far_pairs=int(far.sum()),far_agreement=float((c[ii[far]]==c[jj[far]]).mean()) if far.any() else None,development_pairs=int(dev.sum()),development_agreement=float((c[ii[dev]]==c[jj[dev]]).mean()) if dev.any() else None,test_pairs=int(test.sum()),test_agreement=float((c[ii[test]]==c[jj[test]]).mean()) if test.any() else None,confusion=conf.tolist(),best_color_permutation=[COLORS[j] for j in best],permuted_agreement=sum(int(conf[j,best[j]]) for j in range(3))/len(ii)))
    return sorted(out,key=lambda r:-r['agreement'])

def topology(g,co):
    adj=defaultdict(list)
    for e in g['edges']:
        a,b=e['source'],e['target'];d=tuple(e['delta_ab']);adj[a].append((b,d));adj[b].append((a,tuple(-x for x in d)))
    root=sorted(co)[0];pot={root:(0,0)};q=deque([root]);tree=set()
    while q:
        a=q.popleft()
        for b,d in adj[a]:
            if b in pot:continue
            pot[b]=tuple(pot[a][j]+d[j] for j in range(2));tree.add(frozenset((a,b)));q.append(b)
    cycles=[]
    for e in g['edges']:
        if frozenset((e['source'],e['target'])) in tree:continue
        v=[pot[e['source']][j]+e['delta_ab'][j]-pot[e['target']][j] for j in range(2)]
        cycles.append(dict(source=e['source'],target=e['target'],sum=v))
    nonzero=[r for r in cycles if any(r['sum'])];H=g['H'];mult=[r['sum'][0]//H[0] for r in nonzero];assert all(r['sum']==[m*x for x in H] for r,m in zip(nonzero,mult))
    gcd=math.gcd(*mult);weights=[]
    for x in range(-7,8):
        for y in range(-7,8):
            if sorted([abs(x),abs(y),abs(x-y)])==[1,6,7]:weights.append(dict(weights=[x,y],N=abs(x*H[0]+y*H[1]),direction1_is_difference=abs(x-y)==1))
    return dict(vertices=len(pot),edges=len(g['edges']),cycle_rank=len(cycles),cycle_histogram={str(k):v for k,v in Counter(tuple(r['sum']) for r in cycles).items()},nonzero_cycles=nonzero,generator=[gcd*x for x in H],generator_primitive=math.gcd(*H)==1,integer_neighbor_maps=weights,conclusion='The connected signed-family graph fixes coordinates up to origin modulo H. Its cycle subgroup is generated by H. Any different integral closure requires changed link equations; isolated index changes cannot preserve all edges.')

def domains(ab,c,b,ids):
    # Every rank-two HNF subgroup of index 200..399 containing (16,16).
    # This tests a 2D compression, without assuming A or B string order.
    tr=~np.isin(b,[2,5,8,11]);rows=[]
    for q in [1,2,4,8,16]:
        for p in range(1,400//q+1):
            if not 200<=p*q<=399:continue
            for r in range(p):
                if (16-(16//q)*r)%p:continue
                k=hnf(ab,p,q,r);ct=np.bincount(k[tr]*3+c[tr],minlength=p*q*3).reshape(p*q,3)
                rows.append(dict(p=p,q=q,r=r,area=p*q,development_minority=int(ct.sum()-ct.max(1).sum()),development_occupied=int((ct.sum(1)>0).sum()),physical_kernel_A=bool(hnf(np.array([[6,-7]]),p,q,r)[0]==0),physical_kernel_B=bool(hnf(np.array([[7,-6]]),p,q,r)[0]==0)))
    rows.sort(key=lambda r:(r['development_minority'],r['area']))
    selected=rows[:8]+[r for r in rows if r['physical_kernel_A'] or r['physical_kernel_B']]
    seen=set();details=[]
    for r in selected:
        sig=(r['p'],r['q'],r['r'])
        if sig in seen:continue
        seen.add(sig);k=hnf(ab,*sig);sc,_,_=predict(k,c,b)
        details.append(dict(r,score=sc,**conflicts(k,c,ids),motif=motif(k,c,b,ids,r['area'])))
    return dict(enumerated=len(rows),ranking=rows,selected=details,policy='Contains (16,16) as a hypothesis, not an exact established law. Areas 200..399; development sections exclude 2/5/8/11. New scores are exploratory reuse. Low minority counts alone do not establish an independent transverse period or physical string mapping.')

def staircase(g,obs,co,ids,c,b):
    # Learn rules on development, then assess measured paths on held sections.
    D=[(1,0),(0,1),(-1,1),(-1,0),(0,-1),(1,-1)];ix={o:i for i,o in enumerate(ids)};n=len(ids);nx=np.full((6,n+1),n,int)
    for e in g['edges']:
        a,z=e['source'],e['target']
        if a not in ix or z not in ix:continue
        d=tuple(e['delta_ab'])
        if tuple(np.array(co[z])-co[a])!=d:continue
        j=D.index(d);nx[j,ix[a]]=ix[z];nx[(j+3)%6,ix[z]]=ix[a]
    ranking=[];traces=[]
    for col in [0,1]:
        models=[];allowed=np.r_[np.isin(c,[2,col]),False]
        for cone in range(3):
            for u in range(1,7):
                for v in range(1,7):
                    word=[cone]*u+[(cone+1)%6]*v;starts=np.repeat(np.arange(n),len(word));phase=np.tile(np.arange(len(word)),n);where=starts.copy();active=allowed[where];length=active.astype(int)
                    for j in range(24):
                        dest=nx[np.array(word)[(phase+j)%len(word)],where];active=active&allowed[dest];where=np.where(active,dest,n);length+=active
                    best=length.reshape(n,-1).max(1);dev=~np.isin(b,[2,5,8,11]);row=dict(palette='BR' if col==0 else 'BY',cone=cone,runs=[u,v],development_score=float(np.maximum(best[dev]-5,0).mean()),test_score=float(np.maximum(best[~dev]-5,0).mean()),max_length=int(best.max()))
                    models.append((row,word,length.reshape(n,-1)))
        models.sort(key=lambda r:-r[0]['development_score']);ranking.extend(x[0] for x in models);row,word,length=models[0];used=set()
        for i in np.argsort(-length.max(1),kind='stable'):
            phase=int(length[i].argmax());size=int(length[i,phase])
            if size<8:break
            chain=[int(i)]
            for j in range(size-1):chain.append(int(nx[word[(phase+j)%len(word)],chain[-1]]))
            if len(set(chain)&used)>size//3 or len(set(c[chain]))<2:continue
            used.update(chain);traces.append(dict(palette=row['palette'],cone=row['cone'],runs=row['runs'],ids=[ids[z] for z in chain],word=[D[word[(phase+j)%len(word)]] for j in range(size-1)],length=size))
    edgepairs={frozenset((e['source'],e['target'])) for e in g['edges']};occ={v:o for o,v in co.items() if o in ix}
    for trace in traces:
        examples=[]
        for m in range(-12,13):
            if not m:continue
            pairs=[(o,occ.get((co[o][0]+16*m,co[o][1]+16*m))) for o in trace['ids']];pairs=[(a,z) for a,z in pairs if z and min(abs(int(b[ix[a]])-int(b[ix[z]])),12-abs(int(b[ix[a]])-int(b[ix[z]])))>=2]
            if len(pairs)<6:continue
            links=sum(frozenset((pairs[j][0],pairs[j+1][0])) in edgepairs and frozenset((pairs[j][1],pairs[j+1][1])) in edgepairs for j in range(len(pairs)-1));agree=sum(obs[a]['color']==obs[z]['color'] for a,z in pairs)
            examples.append(dict(multiplier=m,pairs=pairs,agree=agree,n=len(pairs),links=links,complete_measured_trace=len(pairs)==trace['length'] and links==trace['length']-1))
        trace['examples']=examples;trace['repeat_supported']=sum(x['agree']/x['n']>=.85 and x['links']>=4 for x in examples)>=2
    return dict(ranking=ranking,traces=traces,coverage=len({o for r in traces for o in r['ids']}),repeated_coverage=len({o for r in traces if r['repeat_supported'] for o in r['ids']}),policy='Nonunique palette paths. Learned development rules; no exact staircase imposed at branches or unknowns. A complete trace counterpart is still a partial motif, not a complete spiral design.')
